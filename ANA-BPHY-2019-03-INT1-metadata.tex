\AtlasTitle{Study of the rare decays of \Bs and \Bd into muon pairs using data collected between 2015 and 2018 with the ATLAS detector}

% Abstract - % directly after { is important for correct indentation
\AtlasAbstract{%
  This note documents the analysis of the rare \ensuremath{B^0_s\rightarrow\mu\mu}\xspace
  and \ensuremath{B^0_d\rightarrow\mu\mu}\xspace decays using the data collected 
  with the ATLAS Experiment between 2015 and 2018. 
  The analysis strategy and techniques are similar to previous ATLAS results using Run 1 
  and Partial Run 2 data.
}
